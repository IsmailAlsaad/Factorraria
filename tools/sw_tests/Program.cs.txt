using System;
using System.Collections.Generic;
using Factorraria.Common.SuperWeapons;

// Super weapons patch S3.1: checks for Common/SuperWeapons/SuperWeaponAssembly.cs (pure code, no Terraria types).
// They are .txt on purpose: tModLoader compiles every .cs file in the mod folder. See README.txt.

static class Program
{
    static int passed, failed;

    class World
    {
        readonly Dictionary<(int, int), PartCell> cells = new Dictionary<(int, int), PartCell>();

        public PartCell Get(int x, int y) => cells.TryGetValue((x, y), out PartCell c) ? c : default;

        public void Drill(int x, int y, int facing)
        {
            for (int dx = 0; dx < SuperWeaponAssembly.DrillWidth; dx++)
                for (int dy = 0; dy < SuperWeaponAssembly.DrillHeight; dy++)
                    cells[(x + dx, y + dy)] = new PartCell(SuperWeaponPartKind.Drill, facing, dx == 0 && dy == 0);
        }

        public void Thruster(int x, int y, int facing) => cells[(x, y)] = new PartCell(SuperWeaponPartKind.Thruster, facing, true);

        public void Remove(int x, int y) => cells.Remove((x, y));

        public void RemoveDrill(int x, int y)
        {
            for (int dx = 0; dx < SuperWeaponAssembly.DrillWidth; dx++)
                for (int dy = 0; dy < SuperWeaponAssembly.DrillHeight; dy++)
                    cells.Remove((x + dx, y + dy));
        }

        // A complete weapon around a tank whose top-left tile is (tx, ty).
        public static World Weapon(int tx, int ty, int facing)
        {
            var w = new World();
            if (facing > 0)
            {
                w.Drill(tx + 4, ty, 1);
                for (int r = 0; r < 3; r++) w.Thruster(tx - 1, ty + r, 1);
            }
            else
            {
                w.Drill(tx - 2, ty, -1);
                for (int r = 0; r < 3; r++) w.Thruster(tx + 4, ty + r, -1);
            }
            return w;
        }

        public AssemblyResult Eval(int tx, int ty) => SuperWeaponAssembly.Evaluate(tx, ty, Get);
    }

    static void Check(string name, bool ok)
    {
        if (ok) { passed++; Console.WriteLine("pass  " + name); }
        else { failed++; Console.WriteLine("FAIL  " + name); }
    }

    static void CheckIssue(string name, AssemblyResult r, AssemblyIssue issue, int row, int x, int y)
    {
        bool ok = !r.Assembled && r.Issue == issue && r.IssueRow == row && r.IssueX == x && r.IssueY == y;
        Check(name + " [" + r.Issue + " row " + r.IssueRow + " at " + r.IssueX + "," + r.IssueY + "]", ok);
    }

    static int Main()
    {
        const int tx = 100, ty = 50;

        // ---- valid, both directions
        AssemblyResult r = World.Weapon(tx, ty, 1).Eval(tx, ty);
        Check("valid weapon facing right", r.Assembled && r.Facing == 1 && r.Issue == AssemblyIssue.None);
        r = World.Weapon(tx, ty, -1).Eval(tx, ty);
        Check("valid weapon facing left", r.Assembled && r.Facing == -1 && r.Issue == AssemblyIssue.None);

        // ---- empty world
        r = new World().Eval(tx, ty);
        CheckIssue("lone tank: no drill, expected on the right", r, AssemblyIssue.NoDrill, 0, tx + 4, ty);
        Check("lone tank: facing unknown", r.Facing == 0);

        // ---- missing thruster, each row (facing right and left)
        for (int row = 0; row < 3; row++)
        {
            var w = World.Weapon(tx, ty, 1);
            w.Remove(tx - 1, ty + row);
            CheckIssue("right: thruster row " + row + " missing", w.Eval(tx, ty), AssemblyIssue.ThrusterMissing, row, tx - 1, ty + row);

            w = World.Weapon(tx, ty, -1);
            w.Remove(tx + 4, ty + row);
            CheckIssue("left: thruster row " + row + " missing", w.Eval(tx, ty), AssemblyIssue.ThrusterMissing, row, tx + 4, ty + row);
        }

        // ---- no thruster at all: the first row is reported
        {
            var w = World.Weapon(tx, ty, 1);
            for (int row = 0; row < 3; row++) w.Remove(tx - 1, ty + row);
            CheckIssue("right: no thrusters, row 0 reported first", w.Eval(tx, ty), AssemblyIssue.ThrusterMissing, 0, tx - 1, ty);
        }

        // ---- thruster one row off (shifted up / down by one)
        {
            var w = World.Weapon(tx, ty, 1);
            for (int row = 0; row < 3; row++) w.Remove(tx - 1, ty + row);
            for (int row = -1; row <= 1; row++) w.Thruster(tx - 1, ty + row, 1);
            CheckIssue("right: thrusters shifted up by one", w.Eval(tx, ty), AssemblyIssue.ThrusterRowOff, 2, tx - 1, ty + 2);

            w = World.Weapon(tx, ty, 1);
            for (int row = 0; row < 3; row++) w.Remove(tx - 1, ty + row);
            for (int row = 1; row <= 3; row++) w.Thruster(tx - 1, ty + row, 1);
            CheckIssue("right: thrusters shifted down by one", w.Eval(tx, ty), AssemblyIssue.ThrusterRowOff, 0, tx - 1, ty);

            w = World.Weapon(tx, ty, -1);
            for (int row = 0; row < 3; row++) w.Remove(tx + 4, ty + row);
            for (int row = -1; row <= 1; row++) w.Thruster(tx + 4, ty + row, -1);
            CheckIssue("left: thrusters shifted up by one", w.Eval(tx, ty), AssemblyIssue.ThrusterRowOff, 2, tx + 4, ty + 2);
        }

        // ---- thruster in the wrong column (one tile further away) is just missing
        {
            var w = World.Weapon(tx, ty, 1);
            for (int row = 0; row < 3; row++) { w.Remove(tx - 1, ty + row); w.Thruster(tx - 2, ty + row, 1); }
            CheckIssue("right: thrusters one column too far", w.Eval(tx, ty), AssemblyIssue.ThrusterMissing, 0, tx - 1, ty);
        }

        // ---- drill one and two rows off, up and down, on both sides
        foreach (int dy in new[] { -2, -1, 1, 2 })
        {
            var w = World.Weapon(tx, ty, 1);
            w.RemoveDrill(tx + 4, ty);
            w.Drill(tx + 4, ty + dy, 1);
            CheckIssue("right: drill " + dy + " rows off", w.Eval(tx, ty), AssemblyIssue.DrillRowOff, 0, tx + 4, ty + dy);

            w = World.Weapon(tx, ty, -1);
            w.RemoveDrill(tx - 2, ty);
            w.Drill(tx - 2, ty + dy, -1);
            CheckIssue("left: drill " + dy + " rows off", w.Eval(tx, ty), AssemblyIssue.DrillRowOff, 0, tx - 2, ty + dy);
        }

        // ---- drill three rows off does not touch the tank: no drill
        {
            var w = World.Weapon(tx, ty, 1);
            w.RemoveDrill(tx + 4, ty);
            w.Drill(tx + 4, ty + 3, 1);
            r = w.Eval(tx, ty);
            Check("right: drill three rows off does not touch the tank", !r.Assembled && r.Issue == AssemblyIssue.NoDrill);
        }

        // ---- drill one column away from the tank end: no drill
        {
            var w = World.Weapon(tx, ty, 1);
            w.RemoveDrill(tx + 4, ty);
            w.Drill(tx + 5, ty, 1);
            r = w.Eval(tx, ty);
            Check("right: drill one tile away from the tank", !r.Assembled && r.Issue == AssemblyIssue.NoDrill);
        }

        // ---- missing drill: the expected side follows the thrusters
        {
            var w = World.Weapon(tx, ty, 1);
            w.RemoveDrill(tx + 4, ty);
            CheckIssue("right-facing weapon without its drill", w.Eval(tx, ty), AssemblyIssue.NoDrill, 0, tx + 4, ty);

            w = World.Weapon(tx, ty, -1);
            w.RemoveDrill(tx - 2, ty);
            CheckIssue("left-facing weapon without its drill (expected on the left)", w.Eval(tx, ty), AssemblyIssue.NoDrill, 0, tx - 2, ty);
        }

        // ---- facing problems
        {
            var w = World.Weapon(tx, ty, 1);
            w.RemoveDrill(tx + 4, ty);
            w.Drill(tx + 4, ty, -1);
            r = w.Eval(tx, ty);
            CheckIssue("drill on the right facing left", r, AssemblyIssue.DrillWrongFacing, 0, tx + 4, ty);
            Check("wrong-facing drill still reports the side", r.Facing == 1);

            w = World.Weapon(tx, ty, 1);
            w.Thruster(tx - 1, ty + 1, -1);
            CheckIssue("one thruster faces the other way (row 1)", w.Eval(tx, ty), AssemblyIssue.ThrusterWrongFacing, 1, tx - 1, ty + 1);

            w = World.Weapon(tx, ty, -1);
            w.Thruster(tx + 4, ty, 1);
            w.Thruster(tx + 4, ty + 2, 1);
            CheckIssue("two thrusters mixed: the top row is reported first", w.Eval(tx, ty), AssemblyIssue.ThrusterWrongFacing, 0, tx + 4, ty);

            // all three parts facing the wrong way for their side
            w = new World();
            w.Drill(tx + 4, ty, -1);
            for (int row = 0; row < 3; row++) w.Thruster(tx - 1, ty + row, -1);
            CheckIssue("a whole weapon built backwards", w.Eval(tx, ty), AssemblyIssue.DrillWrongFacing, 0, tx + 4, ty);
        }

        // ---- drill on the wrong end (thrusters on one end, drill built on the same-facing but wrong end)
        {
            var w = new World();
            w.Drill(tx - 2, ty, 1);                       // left end, but facing right
            for (int row = 0; row < 3; row++) w.Thruster(tx + 4, ty + row, 1);
            CheckIssue("drill on the left end, parts all facing right", w.Eval(tx, ty), AssemblyIssue.DrillWrongFacing, 0, tx - 2, ty);
        }

        // ---- drills on both ends
        {
            var w = World.Weapon(tx, ty, 1);
            w.Drill(tx - 2, ty, -1);
            for (int row = 0; row < 3; row++) w.Remove(tx - 1, ty + row);   // the left drill overlaps the thruster column
            w.Drill(tx - 2, ty, -1);
            r = w.Eval(tx, ty);
            Check("drills on both ends", !r.Assembled && r.Issue == AssemblyIssue.DrillsOnBothEnds);
        }

        // ---- breaking a part invalidates, putting it back restores
        {
            var w = World.Weapon(tx, ty, 1);
            Check("before breaking: valid", w.Eval(tx, ty).Assembled);
            w.Remove(tx - 1, ty + 1);
            Check("thruster mined: invalid", !w.Eval(tx, ty).Assembled);
            w.Thruster(tx - 1, ty + 1, 1);
            Check("thruster placed back: valid again", w.Eval(tx, ty).Assembled);
            w.RemoveDrill(tx + 4, ty);
            Check("drill mined: invalid", !w.Eval(tx, ty).Assembled);
            w.Drill(tx + 4, ty, 1);
            Check("drill placed back: valid again", w.Eval(tx, ty).Assembled);
        }

        // ---- the same weapon read twice gives the same answer (reload keeps it valid: nothing is stored in the check)
        {
            var w = World.Weapon(tx, ty, -1);
            AssemblyResult a = w.Eval(tx, ty);
            AssemblyResult b = w.Eval(tx, ty);
            Check("repeated evaluation is stable", a.Assembled && b.Assembled && a.Facing == b.Facing);
        }

        // ---- extra parts around the weapon are ignored
        {
            var w = World.Weapon(tx, ty, 1);
            w.Thruster(tx - 1, ty - 1, 1);
            w.Thruster(tx - 1, ty + 3, -1);
            w.Thruster(tx + 10, ty, -1);
            w.Drill(tx + 20, ty + 7, -1);
            Check("stray thrusters and drills elsewhere do not matter", w.Eval(tx, ty).Assembled);
        }

        // ---- a drill tile that is not the origin does not count as the drill
        {
            var w = World.Weapon(tx, ty, 1);
            w.RemoveDrill(tx + 4, ty);
            w.Drill(tx + 3, ty, 1);   // its origin is one tile inside the tank's footprint, its second column lands on tx + 4
            r = w.Eval(tx, ty);
            Check("a drill cell that is not an origin is not a drill", !r.Assembled && r.Issue == AssemblyIssue.NoDrill);
        }

        // ---- a thruster-shaped cell that is not a thruster (a drill cell in the column) is missing
        {
            var w = World.Weapon(tx, ty, 1);
            w.Remove(tx - 1, ty + 2);
            w.Drill(tx - 2, ty + 2, 1);
            r = w.Eval(tx, ty);
            Check("a drill cell in the thruster column is no thruster", !r.Assembled && r.Issue == AssemblyIssue.ThrusterMissing && r.IssueRow == 2);
        }

        // ---- other tank positions, including the world's low coordinates (the reader answers 'nothing' outside the world)
        {
            var w = World.Weapon(2, 0, 1);
            Check("tank near the top-left of the world, right", w.Eval(2, 0).Assembled);
            w = World.Weapon(1, 1, -1);
            Check("tank near the top-left of the world, left", w.Eval(1, 1).Assembled);
            r = new World().Eval(0, 0);
            Check("lone tank at (0,0) does not throw", !r.Assembled && r.Issue == AssemblyIssue.NoDrill);
        }

        // ---- two tanks side by side do not confuse each other when their parts are shared
        {
            // [thrusters][tank A][drill][tank B]... a drill touching tank B's left end is also the right drill of tank A
            var w = new World();
            for (int row = 0; row < 3; row++) w.Thruster(tx - 1, ty + row, 1);
            w.Drill(tx + 4, ty, 1);
            Check("tank A valid", w.Eval(tx, ty).Assembled);
            r = w.Eval(tx + 6, ty);
            Check("tank B (drill on its left end facing right, nothing behind it) is not valid", !r.Assembled);
        }

        // ---- descriptions exist for every state
        {
            Check("describe: assembled right", SuperWeaponAssembly.Describe(true, 1, AssemblyIssue.None, 0).Contains("right"));
            Check("describe: assembled left", SuperWeaponAssembly.Describe(true, -1, AssemblyIssue.None, 0).Contains("left"));
            bool all = true;
            foreach (AssemblyIssue issue in Enum.GetValues(typeof(AssemblyIssue)))
            {
                string text = SuperWeaponAssembly.Describe(false, 0, issue, 1);
                if (string.IsNullOrEmpty(text)) all = false;
            }
            Check("describe: every issue has a text", all);
            Check("describe: thruster rows are shown 1-based", SuperWeaponAssembly.Describe(false, 1, AssemblyIssue.ThrusterMissing, 2).Contains("row 3"));
        }

        // ---- patch S4.1: which weapon fires (SuperWeaponTrigger.Pick)
        {
            var list = new List<TriggerCandidate>();
            Check("trigger: no candidates -> none", SuperWeaponTrigger.Pick(1000f, list, false) == -1);
            Check("trigger: null list -> none", SuperWeaponTrigger.Pick(1000f, null, false) == -1);

            list.Add(new TriggerCandidate(1, 500f, 1));
            Check("trigger: facing right, wall on the right -> fires", SuperWeaponTrigger.Pick(1000f, list, false) == 0);
            Check("trigger: facing right, wall on the left -> does not fire", SuperWeaponTrigger.Pick(100f, list, false) == -1);
            Check("trigger: wall exactly at the weapon's x -> does not fire", SuperWeaponTrigger.Pick(500f, list, false) == -1);
            Check("trigger: busy (another weapon igniting) -> none", SuperWeaponTrigger.Pick(1000f, list, true) == -1);

            list.Clear();
            list.Add(new TriggerCandidate(1, 500f, -1));
            Check("trigger: facing left, wall on the left -> fires", SuperWeaponTrigger.Pick(100f, list, false) == 0);
            Check("trigger: facing left, wall on the right -> does not fire", SuperWeaponTrigger.Pick(900f, list, false) == -1);

            list.Clear();
            list.Add(new TriggerCandidate(1, 100f, 1));    // far
            list.Add(new TriggerCandidate(2, 800f, 1));    // near
            list.Add(new TriggerCandidate(3, 900f, -1));   // nearest but faces away from a wall on its right
            Check("trigger: the nearest weapon that faces the wall fires", SuperWeaponTrigger.Pick(1000f, list, false) == 1);

            list.Clear();
            list.Add(new TriggerCandidate(7, 400f, 1));
            list.Add(new TriggerCandidate(3, 600f, -1));
            Check("trigger: equal distance on both sides goes to the lower id", SuperWeaponTrigger.Pick(500f, list, false) == 1);
            list.Clear();
            list.Add(new TriggerCandidate(9, 400f, 1));
            list.Add(new TriggerCandidate(4, 400f, 1));
            Check("trigger: exact tie (same x) goes to the lower id", SuperWeaponTrigger.Pick(1000f, list, false) == 1);

            list.Clear();
            list.Add(new TriggerCandidate(1, 400f, 0));
            Check("trigger: facing 0 never fires", SuperWeaponTrigger.Pick(1000f, list, false) == -1);

            // two weapons on opposite sides of the wall: each only fires at the wall on its own facing side
            list.Clear();
            list.Add(new TriggerCandidate(1, 100f, 1));    // left of the wall, faces right: fires
            list.Add(new TriggerCandidate(2, 1900f, 1));   // right of the wall, faces right (away): does not
            Check("trigger: a weapon on the far side facing away does not fire", SuperWeaponTrigger.Pick(1000f, list, false) == 0);
        }

        Console.WriteLine();
        Console.WriteLine(failed == 0 ? "ALL PASSED (" + passed + " checks)" : failed + " FAILED, " + passed + " passed");
        return failed == 0 ? 0 : 1;
    }
}
