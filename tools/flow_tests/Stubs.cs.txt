namespace Microsoft.Xna.Framework
{
    public struct Point : IEquatable<Point>
    {
        public int X, Y;
        public Point(int x, int y) { X = x; Y = y; }
        public static Point Zero => new Point(0, 0);
        public static Point operator +(Point a, Point b) => new Point(a.X + b.X, a.Y + b.Y);
        public static Point operator -(Point a, Point b) => new Point(a.X - b.X, a.Y - b.Y);
        public static bool operator ==(Point a, Point b) => a.X == b.X && a.Y == b.Y;
        public static bool operator !=(Point a, Point b) => !(a == b);
        public bool Equals(Point o) => this == o;
        public override bool Equals(object o) => o is Point p && this == p;
        public override int GetHashCode() => X * 73856093 ^ Y * 19349663;
    }
    public struct Color { public static Color Yellow, LightGreen; }
}

namespace Factorraria.Common.Machines
{
    using Factorraria.Common.Liquids;
    using Terraria;
    using Terraria.DataStructures;

    public class BaseMachine : TileEntity
    {
        public bool isOn;
        public int InCount = 1, OutCount = 1;
        LiquidStack[] ins, outs;
        public LiquidStack[] InputLiquids => ins ??= Make(InCount);
        public LiquidStack[] OutputLiquids => outs ??= Make(OutCount);
        static LiquidStack[] Make(int n) { var a = new LiquidStack[n]; for (int i = 0; i < n; i++) a[i] = new LiquidStack(); return a; }
    }
    public class MotorTileEntityBase : BaseMachine
    {
        public float PumpStrength = 5 * 3600f;
        public float PipeEquivalentMaxFlowRate = 20 * 3600f;
        public Direction Facing = Direction.Right;
        public MotorTileEntityBase() { InCount = 0; OutCount = 0; }
    }
    public static class TileEntityHelper
    {
        public static Dictionary<(int, int), TileEntity> Entities = new();
        public static bool TryGetEntityFromTile(int i, int j, out TileEntity entity, out Point16 corner)
        {
            corner = default; entity = null;
            if (!Main.tile[i, j].HasTile) return false;
            return Entities.TryGetValue((i, j), out entity);
        }
    }
}
